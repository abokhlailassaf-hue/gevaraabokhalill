# ===== مرحلة البناء: تثبيت الحزم وبناء الإضافات الأصلية (better-sqlite3) =====
FROM node:20-bookworm-slim AS build

# أدوات البناء المطلوبة فقط لتصريف better-sqlite3 (إضافة native)
RUN apt-get update && apt-get install -y --no-install-recommends \
    python3 \
    make \
    g++ \
  && rm -rf /var/lib/apt/lists/*

WORKDIR /app

COPY package.json package-lock.json* ./
RUN npm install --omit=dev

COPY . .

# ===== المرحلة النهائية: صورة تشغيل نظيفة وخفيفة، بدون أدوات البناء =====
FROM node:20-bookworm-slim

WORKDIR /app

# libatomic مطلوبة وقت التشغيل لبعض إضافات better-sqlite3 على بعض المعماريات
RUN apt-get update && apt-get install -y --no-install-recommends \
    libatomic1 \
  && rm -rf /var/lib/apt/lists/*

COPY --from=build /app /app

ENV NODE_ENV=production

EXPOSE 3000

CMD ["node", "server.js"]
